# Inter-VLAN Routing using SVI (Switch Virtual Interface) Reference

enable
configure terminal
hostname L3-SW

# Enable Layer 3 Routing
ip routing

# VLAN Creation
vlan 10
name HR

vlan 20
name SALES

vlan 30
name DEV

# SVI Configuration (Gateway for VLANs)
interface vlan 10
ip address 192.168.10.1 255.255.255.0
no shutdown

interface vlan 20
ip address 192.168.20.1 255.255.255.0
no shutdown

interface vlan 30
ip address 192.168.30.1 255.255.255.0
no shutdown

# Assign Ports to VLANs
interface fa 0/1
switchport mode access
switchport access vlan 10

interface fa 0/2
switchport mode access
switchport access vlan 20

interface fa 0/3
switchport mode access
switchport access vlan 30


# Verification Commands
show vlan brief
show ip route
ping 192.168.20.1
